// ========================================================= // AUTH CONTEXT (Manages Login, Role, Token) // =========================================================

import { createContext, useState, useEffect } from "react"; export const AuthContext = createContext();

export default function AuthProvider({ children }) { const [user, setUser] = useState(null);

useEffect(() => { const saved = localStorage.getItem("auth"); if (saved) setUser(JSON.parse(saved)); }, []);

const login = (data) => { setUser(data); localStorage.setItem("auth", JSON.stringify(data)); };

const logout = () => { setUser(null); localStorage.removeItem("auth"); };

return ( <AuthContext.Provider value={{ user, login, logout }}> {children} </AuthContext.Provider> ); }

// ========================================================= // AXIOS CLIENT // =========================================================

import axios from "axios"; const api = axios.create({ baseURL: "https://localhost:7100/api" });

api.interceptors.request.use((config) => { const auth = localStorage.getItem("auth"); if (auth) config.headers.Authorization = Bearer ${JSON.parse(auth).token}; return config; });

export default api;

// ADMIN DASHBOARD export default function AdminDashboard() { return <h1>Admin Dashboard</h1>; } // ========================================================= // ADMIN → CREATE BANK (FULL WORKING API INTEGRATION) // ========================================================= import { useState } from "react"; import api from "../../api/axiosClient";

export default function CreateBank() { const [name, setName] = useState(""); const [code, setCode] = useState(""); const [loading, setLoading] = useState(false); const [message, setMessage] = useState("");

const submit = async () => { try { setLoading(true); const body = { name, code }; await api.post("/banks", body); // ASP.NET Core → POST api/banks setMessage("Bank created successfully!"); setName(""); setCode(""); } catch (err) { setMessage("Error: " + err.response?.data || "Unknown error"); } finally { setLoading(false); } };

return ( <div> <h1>Create Bank</h1>

<label>Name</label>
  <input
    type="text"
    value={name}
    onChange={(e) => setName(e.target.value)}
    placeholder="Enter bank name"
  />

  <label>Code</label>
  <input
    type="text"
    value={code}
    onChange={(e) => setCode(e.target.value)}
    placeholder="Enter bank code"
  />

  <button disabled={loading} onClick={submit}>
    {loading ? "Creating..." : "Create Bank"}
  </button>

  {message && <p>{message}</p>}
</div>

); }

import { useContext, useState } from "react"; import api from "../../api/axiosClient"; import { AuthContext } from "../../context/AuthContext";

export default function Login() { const { login } = useContext(AuthContext); const [email, setEmail] = useState(""); const [password, setPassword] = useState("");
